# quakegeneric

The real project quakegeneric of the Rust translator's `tests/manual` ([lookibed/c2das](https://github.com/lookibed/c2das)) converted to daslang by [c2das-re](https://github.com/lookibed/c2das-re), as its head-to-head converts our side. Everything in this directory is generated by `scripts/publish_ports.sh --program quakegeneric` in c2das-re; never edit it, a fix belongs in c2das-re.

## Variants

- `readable/` - for reading and changing the code: Boris, and the weak-model add-a-feature check.
- `fast/` - for running.

`fast/` is converted with `checks: off`: no null or bounds checks, so C bugs are not caught (a null access is a SIGSEGV; an out-of-bounds index reads neighbour memory; AOT keeps its range checks); use `readable/` to debug.

Each variant converts the same program under its own knobs (c2das-re's `ports/variants/`, copied as its `variant.json` with the reason and the numbers behind each knob). x native is the das time over the native build's, above 1 when slower; the lines are the published tree's.

    var spancount : int

            s = bbextents
        } elif (s < 0) {

        }

                    tnext = 8

def D_DrawZSpans(var pspan : espan_s?) {
    var count : int
    var doublecount : int
    var izistep : int
    var izi : int
    var pdest : int16?
    var ltemp : uint
    var zi : double
    var du : float
    var dv : float
    // FIXME: check for clamping/range problems
    // we count on FP exceptions being turned off to avoid range problems
    izistep = int(d_zistepu * 32768.0f * 65536.0f)
    while (true) {
        pdest = d_pzbuffer + (d_zwidth * uint(pspan.v)) + pspan.u
        count = pspan.count
        // calculate the initial 1/z
        du = float(pspan.u)
        dv = float(pspan.v)
        zi = double(d_ziorigin + dv * d_zistepv + du * d_zistepu)
        // we count on FP exceptions being turned off to avoid range problems
        izi = int(zi * 32768.0lf * 65536.0lf)
        if ((int64(intptr(pdest)) & 2l) != 0) {
            var _t0 = pdest
            pdest++
            *_t0 = int16(izi >> 16)
            izi += izistep
            count--
        }
        doublecount = count >> 1
        if (doublecount > 0) {
            while (true) {
                ltemp = uint(izi >> 16)
                izi += izistep
                ltemp |= uint(izi) & 4294901760u
                izi += izistep
                *reinterpret<int?>(pdest) = int(ltemp)
                pdest += 2
                doublecount--
                if (!(doublecount > 0)) {
                    break
                }
            }
        }
        if ((count & 1) != 0) {
            *pdest = int16(izi >> 16)
        }
        pspan = pspan.pnext
        if (!(pspan != null)) {
            break
        }
    }
}
