# plmpeg-stream

The real project plmpeg-stream of the Rust translator's `tests/manual` ([lookibed/c2das](https://github.com/lookibed/c2das)) converted to daslang by [c2das-re](https://github.com/lookibed/c2das-re), as its head-to-head converts our side. Everything in this directory is generated by `scripts/publish_ports.sh --program plmpeg-stream` in c2das-re; never edit it, a fix belongs in c2das-re.

## Variants

- `readable/` - for reading and changing the code: Boris, and the weak-model add-a-feature check.
- `fast/` - for running.

`fast/` is converted with `checks: off`: no null or bounds checks, so C bugs are not caught (a null access is a SIGSEGV; an out-of-bounds index reads neighbour memory; AOT keeps its range checks); use `readable/` to debug.

Each variant converts the same program under its own knobs (c2das-re's `ports/variants/`, copied as its `variant.json` with the reason and the numbers behind each knob). x native is the das time over the native build's, above 1 when slower; the lines are the published tree's.

## Running

The modules require c2das-re's `crt` and `c_runtime` modules: a c2das-re checkout at the commit above, with the daScript its `VERSIONS.md` pins built and `scripts/build_crt.sh` run. The program reads its fixture `fixtures/testsrc2_320x240.m1v` from the Rust repository's project folder; no fixture is copied here (game data and fixtures stay with their own licenses there). With `<c2das-re>` and `<c2das>` the two checkouts:


```das
[c_function]
def plm_video_decode_block(var self : plm_video_t?; _c_block : int) {
    var n : int = 0
    var quant_matrix : uint8?
    // Decode DC coefficient of intra-coded blocks
    if (self.macroblock_intra != 0) {
        var predictor : int
        var dct_size : int
        // DC prediction
        let plane_index : int = _c_block > 3 ? _c_block - 3 : 0
        predictor = self.dc_predictor[plane_index]
        dct_size = int(plm_buffer_read_vlc(self.buffer, plmpeg_file_all__PLM_VIDEO_DCT_SIZE[plane_index]))
        // Read DC coeff
        if (dct_size > 0) {
            let differential : int = plm_buffer_read(self.buffer, dct_size)
            if ((differential & (1 << (dct_size - 1))) != 0) {
                self.block_data[0] = predictor + differential
            } else {
                self.block_data[0] = predictor + (-(1 << dct_size) | (differential + 1))
            }
        } else {
            self.block_data[0] = predictor
        }
        // Save predictor value
        self.dc_predictor[plane_index] = self.block_data[0]
        // Dequantize + premultiply
        self.block_data[0] <<= 3 + 5
        quant_matrix = addr(self.intra_quant_matrix[0])
        n = 1
    } else {
        quant_matrix = addr(self.non_intra_quant_matrix[0])
    }
    // Decode AC coefficients (+DC for non-intra)
    var level : int = 0
    while (TRUE != 0) {
        var run : int = 0
        let coeff : uint16 = plm_buffer_read_vlc_uint(self.buffer, addr(plmpeg_file_all__PLM_VIDEO_DCT_COEFF[0]))
        if ((coeff == 1) && (n > 0) && (plm_buffer_read(self.buffer, 1) == 0)) {
            // end_of_block
            break
        }
        if (coeff == 65535) {
            // escape
            run = plm_buffer_read(self.buffer, 6)
            level = plm_buffer_read(self.buffer, 8)
            if (level == 0) {
                level = plm_buffer_read(self.buffer, 8)
            } elif (level == 128) {
                level = plm_buffer_read(self.buffer, 8) - 256
            } elif (level > 128) {
                level -= 256
            }
        } else {
            run = int(coeff) >> 8
            level = int(coeff) & 255
            if (plm_buffer_read(self.buffer, 1) != 0) {
                level = -level
            }
        }
        n += run
        if (n < 0 || n >= 64) {
            // invalid
            return
        }
        let de_zig_zagged : int = int(plmpeg_file_all__PLM_VIDEO_ZIG_ZAG[n])
        n++
        // Dequantize, oddify, clip
        level = int(uint(level) << 1u)
        if (self.macroblock_intra == 0) {
            level += level < 0 ? -1 : 1
        }
        level = (level * self.quantizer_scale * int(quant_matrix[de_zig_zagged])) >> 4
        if ((level & 1) == 0) {
            level -= level > 0 ? 1 : -1
        }
        if (level > 2047) {
            level = 2047
        } elif (level < -2048) {
            level = -2048
        }
        // Save premultiplied coefficient
        self.block_data[de_zig_zagged] = level * int(plmpeg_file_all__PLM_VIDEO_PREMULTIPLIER_MATRIX[de_zig_zagged])
    }
    // Move block to its place
    var d : uint8?
    var dw : int
    var di : int
    if (_c_block < 4) {
        d = self.frame_current.y.data
        dw = self.luma_width
        di = (self.mb_row * self.luma_width + self.mb_col) << 4
        if ((_c_block & 1) != 0) {
            di += 8
        }
        if ((_c_block & 2) != 0) {
            di += self.luma_width << 3
        }
    } else {
        d = (_c_block == 4) ? self.frame_current.cb.data : self.frame_current.cr.data
        dw = self.chroma_width
        di = ((self.mb_row * self.luma_width) << 2) + (self.mb_col << 3)
    }
    var s : int? = addr(self.block_data[0])
    var si : int = 0
    if (self.macroblock_intra != 0) {
        // Overwrite (no prediction)
        if (n == 1) {
            let clamped : int = int(plmpeg_file_all__plm_clamp((s[0] + 128) >> 8))
            {
                let dest_scan : int = dw - 8
                let source_scan : int = 8 - 8
                for (y in range(8)) {
                    for (x in range(8)) {
                        d[di] = uint8(clamped)
                        si++
                        di++
                    }
                    si += source_scan
                    di += dest_scan
                }
            }
            s[0] = 0
        } else {
            plm_video_idct(s)
            {
                let dest_scan : int = dw - 8
                let source_scan : int = 8 - 8
                for (y in range(8)) {
                    for (x in range(8)) {
                        d[di] = plmpeg_file_all__plm_clamp(s[si])
                        si++
                        di++
                    }
                    si += source_scan
                    di += dest_scan
                }
            }
            crt::memset(reinterpret<void?>(addr(self.block_data[0])), 0, uint64(typeinfo sizeof(type<int[64]>)))
        }
    } else {
        // Add data to the predicted macroblock
        if (n == 1) {
            let value : int = (s[0] + 128) >> 8
            {
                let dest_scan : int = dw - 8
                let source_scan : int = 8 - 8
                for (y in range(8)) {
                    for (x in range(8)) {
                        d[di] = plmpeg_file_all__plm_clamp(int(d[di]) + value)
                        si++
                        di++
                    }
                    si += source_scan
                    di += dest_scan
                }
            }
            s[0] = 0
        } else {
            plm_video_idct(s)
            {
                let dest_scan : int = dw - 8
                let source_scan : int = 8 - 8
                for (y in range(8)) {
                    for (x in range(8)) {
                        d[di] = plmpeg_file_all__plm_clamp(int(d[di]) + s[si])
                        si++
                        di++
                    }
                    si += source_scan
                    di += dest_scan
                }
            }
            crt::memset(reinterpret<void?>(addr(self.block_data[0])), 0, uint64(typeinfo sizeof(type<int[64]>)))
        }
    }
}
```
